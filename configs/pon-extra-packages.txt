# build-packages-extra.yml 的 C 类编译清单
# 格式：<构建路径>|<apk 包名>|<required 或 optional>
# 路径不存在时工作流会按目录名在 package/feeds 下回退查找，找不到才判定缺失。
# mihomo-meta / mihomo-alpha 由 inputs.mihomo 门控（互 PROVIDES/CONFLICTS，只能选一个）。

# ---- iptv（kmod-iptv + luci-app-iptv + i18n）已整体移交固件流程 fw-build.yml，
#      作为独立 apk 随 kmods-<label>-r<run> 预发布交付，本流程不再涉及 iptv。

# ---- 组播代理：omcproxy 是 OpenWrt 内核树自带的核心包 package/network/services/omcproxy
#      （本仓 master 里就有，含 Makefile + files/omcproxy.{config,init}），不在 packages feed；
#      旧仓 build-package run 47 就是这样编成功的（omcproxy-2026.03.07~3abb601a，ninja 9/9 link）。
package/network/services/omcproxy|omcproxy|required
package/feeds/luci/luci-app-omcproxy|luci-app-omcproxy|required

# ---- nikki / mihomo（源码 feeds/nikki）----
package/feeds/nikki/nikki|nikki|required
package/feeds/nikki/luci-app-nikki|luci-app-nikki|required
# luci-i18n-nikki-zh-cn 不是独立源码目录：luci.mk 从 luci-app-nikki/po/zh_Hans 生成
# （LUCI_LC_ALIAS.zh_Hans=zh-cn），所以路径写父包、编父包时一并产出，required 保证没产出就报错。
package/feeds/nikki/luci-app-nikki|luci-i18n-nikki-zh-cn|required
package/feeds/nikki/mihomo-meta|mihomo-meta|required
package/feeds/nikki/mihomo-alpha|mihomo-alpha|required

# ---- passwall2 + 其唯一核心 xray-core ----
package/feeds/passwall2/luci-app-passwall2|luci-app-passwall2|required
package/feeds/packages/xray-core|xray-core|required

# ---- homeproxy（immortalwrt/homeproxy，依赖 sing-box，树内已选）----
package/feeds/homeproxy/luci-app-homeproxy|luci-app-homeproxy|required

# ---- Windows 文件共享：ksmbd（备用，与 samba4 不冲突；默认不装，迁移时再切）----
package/feeds/packages/ksmbd-tools|ksmbd-tools|required
package/feeds/luci/luci-app-ksmbd|luci-app-ksmbd|required

# ---- OpenAppFilter（内核侧 kmod-oaf 由固件流编，这里只编用户态）----
# appfilter 不进包流：它的 Makefile 第 40 行用 $(KERNEL_MAKE) 在同一目录编 oaf.ko，
# 属于「需要内核源码树」的包（incremental 的 preflight 会直接拒绝）。改由固件流编，
# 见 configs/features/kmod-repo.config 的 CONFIG_PACKAGE_appfilter=m。
package/feeds/oaf/luci-app-oaf|luci-app-oaf|required

# ---- OpenClash / SQM：不进固件镜像，只出独立 apk
#      注意：包流只认这份清单（configs/pon-extra/90-c-class.config 并未被任何工作流引用），
#      所以这里的行才是真正决定「编不编」的地方。
#      内核侧依赖（kmod-tun/kmod-nft-tproxy/kmod-inet-diag/kmod-sched-cake/kmod-ifb）由固件流编出，
#      本流程只编应用层，不碰内核源码树。
package/feeds/luci/applications/luci-app-openclash|luci-app-openclash|required
package/feeds/packages/utils/bash|bash|required
# ruby-yaml 不是独立源码目录，由 lang/ruby 的 stdlib 子包产出，路径写父包
package/feeds/packages/lang/ruby|ruby|required
package/feeds/packages/lang/ruby|ruby-yaml|required
package/feeds/packages/utils/unzip|unzip|required
package/feeds/luci/applications/luci-app-sqm|luci-app-sqm|required
# luci-i18n-sqm-zh-cn 由 luci-app-sqm/po/zh_Hans 生成，同样写父包路径
package/feeds/luci/applications/luci-app-sqm|luci-i18n-sqm-zh-cn|required
package/feeds/packages/net/sqm-scripts|sqm-scripts|required
# tc 是虚拟名，实际产物 tc-tiny；nftables 由 nftables-json 提供 —— 二者都在内核树内
package/network/utils/iproute2|tc-tiny|required
package/network/utils/nftables|nftables-json|required
